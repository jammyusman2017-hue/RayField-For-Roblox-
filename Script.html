-- 🌍 MOVEMENT & VISION TAB (ESP & FLY)
-- ==========================================
MainTab:CreateSection("Visual Enhancements")

-- Simple Highlight ESP
MainTab:CreateToggle({
Name = "Player Highlight ESP",
CurrentValue = false,
Callback = function(Value)
_G.EspEnabled = Value
while _G.EspEnabled do
for _, player in pairs(game.Players:GetPlayers()) do
if player ~= LocalPlayer and player.Character then
local highlight = player.Character:FindFirstChild("HubESP") or Instance.new("Highlight")
highlight.Name = "HubESP"
highlight.FillColor = Color3.fromRGB(255, 0, 0)
highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
highlight.Parent = player.Character
highlight.Enabled = true
end
end
task.wait(2)
end
-- Clean up ESP when turned off
if not Value then
for _, player in pairs(game.Players:GetPlayers()) do
if player.Character and player.Character:FindFirstChild("HubESP") then
player.Character.HubESP:Destroy()
end
end
end
end,
})

MainTab:CreateSection("Movement Physics")

-- Basic Fly Mechanic
MainTab:CreateToggle({
Name = "Toggle Fly",
CurrentValue = false,
Callback = function(Value)
Flying = Value
local Character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
local HumanoidRootPart = Character:WaitForChild("HumanoidRootPart")

if Flying then
local BodyVelocity = Instance.new("BodyVelocity")
BodyVelocity.Name = "FlyVelocity"
BodyVelocity.Velocity = Vector3.new(0, 0, 0)
BodyVelocity.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
BodyVelocity.Parent = HumanoidRootPart

task.spawn(function()
while Flying and Character and HumanoidRootPart do
local MoveDirection = Character:WaitForChild("Humanoid").MoveDirection
BodyVelocity.Velocity = MoveDirection * FlySpeed + Vector3.new(0, 0.1, 0)
task.wait()
end
if BodyVelocity then BodyVelocity:Destroy() end
end)
else
if HumanoidRootPart:FindFirstChild("FlyVelocity") then
HumanoidRootPart.FlyVelocity:Destroy()
end
end
end,
})

AdminTab:CreateSection("Data & Stat Modification")

-- 1. Scan the game's leaderstats folder automatically to pull names
local AvailableStatsList = {}
local leaderstats = LocalPlayer:FindFirstChild("leaderstats") or LocalPlayer:FindFirstChild("Leaderstats")

if leaderstats then
for _, statObject in pairs(leaderstats:GetChildren()) do
if statObject:IsA("ValueBase") then -- Finds IntValue, StringValue, NumberValue, etc.
table.insert(AvailableStatsList, statObject.Name)
end
end
end

-- Fallback option if leaderstats array is empty or not yet loaded
if #AvailableStatsList == 0 then
AvailableStatsList = {"Cash", "Coins", "Points"}
end

-- 2. Create the Rayfield Dropdown element containing the scanned stats
local StatDropdown = AdminTab:CreateDropdown({
Name = "Select Game Stat to Modify",
Options = AvailableStatsList,
CurrentOption = AvailableStatsList[1] or "None Chosen",
MultipleOptions = false,
Callback = function(Option)
-- Handles standard string return values cleanly
if type(Option) == "table" then
SelectedStatFromDropdown = Option[1]
else
SelectedStatFromDropdown = Option
end
Rayfield:Notify({Title = "Stat Selected", Content = "Target changed to: " .. tostring(SelectedStatFromDropdown), Duration = 2})
end,
})

-- 3. Input for the new number value
AdminTab:CreateInput({
Name = "New Stat Value",
PlaceholderText = "999999",
ClearTextOnFocus = false,
Callback = function(Text)
StatValueInput = tonumber(Text) or 0
end,
})

-- 4. Execute button applying changes directly to your selected choice
AdminTab:CreateButton({
Name = "Apply Mod to Selected Stat",
Callback = function()
if not IsAuthorized then
return Rayfield:Notify({Title = "Locked", Content = "Please enter the password first.", Duration = 3})
end

if SelectedStatFromDropdown == "" or SelectedStatFromDropdown == "None Chosen" then
return Rayfield:Notify({Title = "Error", Content = "Please pick a stat from the dropdown first.", Duration = 3})
end

local activeLeaderstats = LocalPlayer:FindFirstChild("leaderstats") or LocalPlayer:FindFirstChild("Leaderstats")
if activeLeaderstats then
local finalTargetStat = activeLeaderstats:FindFirstChild(SelectedStatFromDropdown)
if finalTargetStat then
finalTargetStat.Value = StatValueInput
Rayfield:Notify({Title = "Success", Content = SelectedStatFromDropdown .. " adjusted to " .. StatValueInput, Duration = 3})
else
Rayfield:Notify({Title = "Stat Error", Content = "Stat structure moved or missing.", Duration = 3})
end
else
Rayfield:Notify({Title = "Folder Error", Content = "No 'leaderstats' folder active on your player structure.", Duration = 3})
end
end,
})
